async function linhasFiltradas(nivel,ano=STATE.ano){const rows=await carregarNivelAno(nivel,ano);if(nivel==="ESCOLA"){let out=rows.filter(filtroBase);if(STATE.rede==="T"&&STATE.local==="T")return out;const dim=await carregarEscolaFiltros(ano);return out.filter(r=>{const d=dim.get(String(r.id));if(!d)return false;return (STATE.rede==="T"||d.r===STATE.rede)&&(STATE.local==="T"||d.l===STATE.local)})}return rows.filter(r=>filtroBase(r)&&r.r===STATE.rede&&r.l===STATE.local)}
async function registroRS(ano=STATE.ano){const rows=await carregarNivelAno("RS",ano);return rows.find(r=>filtroBase(r)&&r.r===STATE.rede&&r.l===STATE.local)??null}
async function registroSelecionado(ano=STATE.ano){if(STATE.tab==="estado")return registroRS(ano);const nivel=nivelDaAba();const id=idSelecionado();if(id===null)return null;const rows=await linhasFiltradas(nivel,ano);return rows.find(r=>String(r.id)===String(id))??null}

function quantis(values){const a=values.filter(v=>Number.isFinite(v)).sort((x,y)=>x-y);if(!a.length)return[];const q=p=>{const pos=(a.length-1)*p,lo=Math.floor(pos),hi=Math.ceil(pos);return lo===hi?a[lo]:a[lo]+(a[hi]-a[lo])*(pos-lo)};return [q(.2),q(.4),q(.6),q(.8)]}
function corValor(v,qs,pal){if(!Number.isFinite(v))return CORES.semDados;if(v<=qs[0])return pal[0];if(v<=qs[1])return pal[1];if(v<=qs[2])return pal[2];if(v<=qs[3])return pal[3];return pal[4]}
function legendaHTML(qs,pal){const f=x=>CONFIG.indicadores[STATE.indicador].pct?`${fmtNum(x,0)}%`:fmtNum(x,0);if(qs.length<4)return'<div class="legend-row"><span class="legend-cor" style="background:#d8dde1"></span>Sem resultado</div>';const labs=[`≤ ${f(qs[0])}`,`${f(qs[0])}–${f(qs[1])}`,`${f(qs[1])}–${f(qs[2])}`,`${f(qs[2])}–${f(qs[3])}`,`> ${f(qs[3])}`];return labs.map((x,i)=>`<div class="legend-row"><span class="legend-cor" style="background:${pal[i]}"></span>${x}</div>`).join("")+`<div class="legend-row"><span class="legend-cor" style="background:${CORES.semDados}"></span>Sem resultado</div>`}

function initMap(){MAP=L.map("mapa-principal",{preferCanvas:true,zoomControl:true}).setView([-30.15,-53.2],6);L.tileLayer("https://{s}.tile.openstreetmap.de/{z}/{x}/{y}.png",{attribution:'&copy; colaboradores do OpenStreetMap',maxZoom:12,subdomains:"abc"}).addTo(MAP);MAP_LEGEND=L.control({position:"bottomright"});MAP_LEGEND.onAdd=()=>{const d=L.DomUtil.create("div","map-legend");d.innerHTML="";return d};MAP_LEGEND.addTo(MAP)}
function featureId(tab,feature){const p=feature.properties||{};if(tab==="cre")return String(p.CD_REGIONAL);if(tab==="escola")return String(p.e);return String(p.CD_MUNICIPIO)}
function featureNome(tab,feature){const p=feature.properties||{};if(tab==="cre")return p.NM_REGIONAL??nomeNivel("CRE",p.CD_REGIONAL);if(tab==="escola")return nomeNivel("ESCOLA",p.e);return p.NM_MUNICIPIO??nomeNivel("MUNICIPIO",p.CD_MUNICIPIO)}
function geoDaAba(){return STATE.tab==="cre"?CACHE.mapas.cre:STATE.tab==="escola"?CACHE.mapas.escolas:CACHE.mapas.municipios}

async function atualizarMapa(spatialRows){const geo=geoDaAba();const dataMap=new Map(spatialRows.map(r=>[String(r.id),r]));const vals=spatialRows.map(r=>Number(r[STATE.indicador])).filter(Number.isFinite);const qs=quantis(vals),pal=CONFIG.indicadores[STATE.indicador].palette;MAP_FEATURES.clear();if(MAP_LAYER)MAP.removeLayer(MAP_LAYER);
  const stylePoly=f=>{const id=featureId(STATE.tab,f),r=dataMap.get(id),v=r?Number(r[STATE.indicador]):NaN,sel=STATE.tab!=="estado"&&String(idSelecionado())===id;return{color:sel?CORES.selecionado:"#667784",weight:sel?2.5:.7,opacity:.95,fillColor:corValor(v,qs,pal),fillOpacity:Number.isFinite(v)?.78:.28}}
  const opts={style:STATE.tab==="escola"?undefined:stylePoly,onEachFeature:(f,l)=>{const id=featureId(STATE.tab,f),r=dataMap.get(id),nome=featureNome(STATE.tab,f),v=r?Number(r[STATE.indicador]):NaN;MAP_FEATURES.set(id,l);l.bindTooltip(`<strong>${esc(nome)}</strong><br>${esc(CONFIG.indicadores[STATE.indicador].label)}: ${fmtIndicador(v)}`,{sticky:true});l.on("click",()=>onTerritorioClick(id,l))}};
  if(STATE.tab==="escola")opts.pointToLayer=(f,latlng)=>{const id=featureId(STATE.tab,f),r=dataMap.get(id),v=r?Number(r[STATE.indicador]):NaN,sel=String(idSelecionado())===id;return L.circleMarker(latlng,{radius:sel?7:4,color:sel?CORES.selecionado:"#ffffff",weight:sel?2:1,fillColor:corValor(v,qs,pal),fillOpacity:Number.isFinite(v)?.88:.22})};
  MAP_LAYER=L.geoJSON(geo,opts).addTo(MAP);
  if(MAP_LAYER.getBounds&&MAP_LAYER.getBounds().isValid())MAP.fitBounds(MAP_LAYER.getBounds(),{padding:[8,8],maxZoom:STATE.tab==="escola"?8:7});
  const el=MAP_LEGEND.getContainer();if(el)el.innerHTML=`<strong>${esc(CONFIG.indicadores[STATE.indicador].label)}</strong>${legendaHTML(qs,pal)}`;
}
