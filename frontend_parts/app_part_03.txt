function destacarMapaSelecionado(){if(!MAP_LAYER)return;MAP_LAYER.eachLayer(l=>{const f=l.feature;if(!f)return;const id=featureId(STATE.tab,f),sel=String(idSelecionado())===id;if(STATE.tab==="escola"){l.setStyle({radius:sel?7:4,color:sel?CORES.selecionado:"#ffffff",weight:sel?2:1})}else{l.setStyle({color:sel?CORES.selecionado:"#667784",weight:sel?2.5:.7})}})}

function rankingSecundario(r,nivel){if(nivel==="CRE")return `${fmtInt(r.nes)} escolas · ${fmtInt(r.nmu)} municípios`;if(nivel==="MUNICIPIO")return `${fmtInt(r.nes)} escolas`;if(nivel==="ESCOLA"){const d=CACHE.detalhes.escola.get(String(r.id));return d?`${CACHE.nomes.municipio.get(String(d.CD_MUNICIPIO))??""}`:"Escola"}return""}
function renderRanking(rows,nivel){ultimoRanking=rows.slice().filter(r=>Number.isFinite(Number(r[STATE.indicador]))).sort((a,b)=>Number(b[STATE.indicador])-Number(a[STATE.indicador]));const busca=norm(STATE.busca);const vis=busca?ultimoRanking.filter(r=>norm(nomeNivel(nivel,r.id)).includes(busca)):ultimoRanking;const box=document.getElementById("ranking-lista");document.getElementById("ranking-contagem").textContent=`${vis.length.toLocaleString("pt-BR")} com resultado`;if(!vis.length){box.innerHTML='<div class="ranking-vazio">Nenhum resultado para os filtros atuais.</div>';return}box.innerHTML=vis.map((r,i)=>{const id=String(r.id),ativo=STATE.tab!=="estado"&&String(idSelecionado())===id;return `<div class="ranking-item ${ativo?"ativo":""}" data-id="${esc(id)}"><span class="ranking-pos">${i+1}º</span><div class="ranking-nome"><strong>${esc(nomeNivel(nivel,id))}</strong><span>${esc(rankingSecundario(r,nivel))}</span></div><span class="ranking-valor">${fmtIndicador(Number(r[STATE.indicador]))}</span></div>`}).join("");box.querySelectorAll(".ranking-item").forEach(el=>el.addEventListener("click",async()=>{const id=el.dataset.id;if(STATE.tab==="estado"){STATE.tab="municipio";STATE.selecionado.municipio=id;document.querySelectorAll(".aba").forEach(b=>b.classList.toggle("ativa",b.dataset.aba==="municipio"));await atualizarTudo()}else{setSelecionado(id);await atualizarSelecaoSemRecarregarMapa();destacarMapaSelecionado();renderRanking(rows,nivel);const layer=MAP_FEATURES.get(id);if(layer){try{const b=layer.getBounds?layer.getBounds():L.latLngBounds([layer.getLatLng()]);MAP.flyToBounds(b.pad?.(.3)??b,{maxZoom:STATE.tab==="escola"?10:8,duration:.45})}catch(_){}}}}))}
function scrollRankingSelecionado(){const el=document.querySelector('.ranking-item.ativo');if(el)el.scrollIntoView({block:"nearest"})}

function contextoTab(){if(STATE.tab==="estado")return{kick:"Abrangência estadual",titulo:"Rio Grande do Sul",mapa:"Distribuição municipal dos resultados",rank:"Ranking municipal — visão estadual",nivelRank:"MUNICIPIO"};if(STATE.tab==="cre")return{kick:"Abrangência regional",titulo:idSelecionado()?nomeNivel("CRE",idSelecionado()):"Coordenadorias Regionais de Educação",mapa:"30 Coordenadorias Regionais de Educação",rank:"Ranking das CREs",nivelRank:"CRE"};if(STATE.tab==="municipio")return{kick:"Abrangência municipal",titulo:idSelecionado()?nomeNivel("MUNICIPIO",idSelecionado()):"Municípios do Rio Grande do Sul",mapa:"497 municípios",rank:"Ranking dos municípios",nivelRank:"MUNICIPIO"};return{kick:"Abrangência escolar",titulo:idSelecionado()?nomeNivel("ESCOLA",idSelecionado()):"Escolas",mapa:"5.176 escolas georreferenciadas",rank:"Ranking das escolas",nivelRank:"ESCOLA"}}
function atualizarContexto(){const c=contextoTab();document.getElementById("contexto-kicker").textContent=c.kick;document.getElementById("contexto-titulo").textContent=c.titulo;document.getElementById("contexto-subtitulo").textContent=`${STATE.ano} · ${CONFIG.disciplinas[STATE.disciplina]} · ${CONFIG.etapas[STATE.etapa]} · ${CONFIG.redes[STATE.rede]} · ${CONFIG.locais[STATE.local]}`;document.getElementById("mapa-titulo").textContent=c.mapa;document.getElementById("ranking-titulo").textContent=c.rank;document.getElementById("ranking-busca").placeholder=STATE.tab==="escola"?"Buscar escola...":STATE.tab==="cre"?"Buscar CRE...":"Buscar município..."}
